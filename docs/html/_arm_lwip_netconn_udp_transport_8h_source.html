61 int Create(Type type, const char* addr, uint16_t port)
62 {
63 const std::lock_guard<dmq::RecursiveMutex> lock(m_mutex);
64 m_type = type;

120 virtual int Send(xostringstream& os, const DmqHeader& header) override
121 {
122 const std::lock_guard<dmq::RecursiveMutex> lock(m_mutex);
123 if (os.bad() || os.fail() || !m_conn) {
124 return -1;
125 }
126
127 // Only SUBs should block non-ACK traffic
128 if (m_type == Type::SUB && header.GetId() != dmq::ACK_REMOTE_ID) {
129 return -1;
130 }
131
132 if (m_sendTransport != this) {
133 return -1;
134 }
135
136 // Get payload data without string copy if possible
137 auto payload = os.str();
138 uint16_t payloadLen = static_cast<uint16_t>(payload.length());
139
140 if (payloadLen > (BUFFER_SIZE - DmqHeader::HEADER_SIZE)) {
141 return -1;
142 }
143
144 // Prepare header values in Network Byte Order
145 uint16_t marker = htons(header.GetMarker());
146 uint16_t id = htons(header.GetId());
147 uint16_t seqNum = htons(header.GetSeqNum());
148 uint16_t length = htons(payloadLen);
149
150 // Copy Header & Payload into the linear buffer
151 memcpy(m_sendBuffer, &marker, 2);
152 memcpy(m_sendBuffer + 2, &id, 2);
153 memcpy(m_sendBuffer + 4, &seqNum, 2);
154 memcpy(m_sendBuffer + 6, &length, 2);
155
156 if (payloadLen > 0) {
157 memcpy(m_sendBuffer + DmqHeader::HEADER_SIZE, payload.data(), payloadLen);
158 }
159
160 // Allocate a Netbuf
161 struct netbuf* buf = netbuf_new();
162 if (!buf) return -1;
163
164 void* dataPtr = netbuf_alloc(buf, DmqHeader::HEADER_SIZE + payloadLen);
165 if (!dataPtr) {
166 netbuf_delete(buf);
167 return -1;
168 }
169
170 // Copy to NetX/LwIP internal buffer
171 memcpy(dataPtr, m_sendBuffer, DmqHeader::HEADER_SIZE + payloadLen);
172
173 // PUB uses pre-configured IP; SUB uses last-received IP (Reply)
174 err_t err = netconn_sendto(m_conn, buf, &m_remoteIp, m_remotePort);
175
176 netbuf_delete(buf);
177
178 if (err != ERR_OK) return -1;
179
180 return 0;
181 }

183 virtual int Receive(xstringstream& is, DmqHeader& header) override
184 {
185 const std::lock_guard<dmq::RecursiveMutex> lock(m_mutex);
186 if (m_recvTransport != this || !m_conn) {
187 return -1;
188 }
189
190 struct netbuf* buf = nullptr;
191 err_t err = netconn_recv(m_conn, &buf);
192
193 if (err == ERR_TIMEOUT) return -1;
194 if (err != ERR_OK || buf == nullptr) return -1;
195
196 // Capture Sender Info (for ACKs/Replies)
197 ip_addr_t* addr = netbuf_fromaddr(buf);
198 uint16_t port = netbuf_fromport(buf);
199 if (addr) ip_addr_copy(m_remoteIp, *addr);
200 m_remotePort = port;
201
202 uint16_t len = netbuf_len(buf);
203 if (len < DmqHeader::HEADER_SIZE) {
204 netbuf_delete(buf);
205 return -1;
206 }
207
208 // Read Header (Network Byte Order)
209 char headerBuf[DmqHeader::HEADER_SIZE];
210 netbuf_copy(buf, headerBuf, DmqHeader::HEADER_SIZE);
211
212 uint16_t marker, id, seqNum, length;
213 memcpy(&marker, headerBuf, 2);
214 memcpy(&id, headerBuf + 2, 2);
215 memcpy(&seqNum, headerBuf + 4, 2);
216 memcpy(&length, headerBuf + 6, 2);
217
218 header.SetMarker(ntohs(marker));
219 header.SetId(ntohs(id));
220 header.SetSeqNum(ntohs(seqNum));
221 header.SetLength(ntohs(length));
222
223 if (header.GetMarker() != DmqHeader::MARKER || len < DmqHeader::HEADER_SIZE + header.GetLength()) {
224 netbuf_delete(buf);
225 return -1;
226 }
227
228 // Extract Payload
229 uint16_t payloadSize = header.GetLength();
230 if (payloadSize > 0) {
231 if (payloadSize > BUFFER_SIZE) payloadSize = BUFFER_SIZE;
232 netbuf_copy_partial(buf, m_recvBuffer, payloadSize, DmqHeader::HEADER_SIZE);
233 is.clear();
234 is.str("");
235 is.write(m_recvBuffer, payloadSize);
236 }
237
238 netbuf_delete(buf);
239
240 // Handle Reliability Logic
241 if (header.GetId() == dmq::ACK_REMOTE_ID) {
242 if (m_transportMonitor) m_transportMonitor->Remove(header.GetSeqNum());
243 }
244 else if (m_transportMonitor && m_sendTransport) {
245 // Auto-ACK using Send logic
246 dmq::xostringstream ss_ack;
247 DmqHeader ack;
248 ack.SetId(dmq::ACK_REMOTE_ID);
249 ack.SetSeqNum(header.GetSeqNum());
250 ack.SetLength(0);
251 m_sendTransport->Send(ss_ack, ack);
252 }
253
254 return 0;
255 }
