79 int Create(Type type, const char* addr_str, uint16_t port)
80 {
81 tx_mutex_get(&m_mutex, TX_WAIT_FOREVER);
82
83 m_type = type;
84 m_port = port;
85
86 // Parse IP Address
87 m_targetIp = 0;
88 if (addr_str) {
89 uint32_t a, b, c, d;
90 if (std::sscanf(addr_str, "%u.%u.%u.%u", &a, &b, &c, &d) == 4) {
91 // Construct IP ULONG (Assumes NetX expects Host Byte Order for setup)
92 m_targetIp = (a << 24) | (b << 16) | (c << 8) | d;
93 }
94 }

96 // Create the UDP Socket

98 NX_IP_NORMAL, NX_FRAGMENT_OKAY, 0x80, 5);
99 if (ret != NX_SUCCESS) {
100 tx_mutex_put(&m_mutex);
101 return -1;
102 }
103
104 // Bind
105 if (type == Type::SUB) {
106 ret = nx_udp_socket_bind(&m_socket, port, NX_WAIT_FOREVER);
107 } else {
108 ret = nx_udp_socket_bind(&m_socket, NX_ANY_PORT, NX_WAIT_FOREVER);
109 }
110

112 nx_udp_socket_delete(&m_socket);

144 virtual int Send(xostringstream& os, const DmqHeader& header) override
145 {
146 tx_mutex_get(&m_mutex, TX_WAIT_FOREVER);
147
148 if (os.bad() || os.fail()) {
149 tx_mutex_put(&m_mutex);
150 return -1;
151 }
152
153 // Block regular data on SUB socket (only ACKs allowed)
154 if (m_type == Type::SUB && header.GetId() != dmq::ACK_REMOTE_ID) {
155 tx_mutex_put(&m_mutex);
156 return -1;
157 }
158
159 if (m_sendTransport != this) {
160 tx_mutex_put(&m_mutex);
161 return -1;
162 }
163
164 auto payload = os.str();
165 if (payload.length() > UINT16_MAX) {
166 tx_mutex_put(&m_mutex);
167 return -1;
168 }
169
170 // 1. Allocate NetX Packet
171 NX_PACKET* packet_ptr = nullptr;
172 UINT ret = nx_packet_allocate(m_pool, &packet_ptr, NX_UDP_PACKET, NX_NO_WAIT);
173 if (ret != NX_SUCCESS) {
174 tx_mutex_put(&m_mutex);
175 return -1;
176 }
177
178 // 2. Prepare Header (Network Byte Order)
179 DmqHeader headerCopy = header;
180 headerCopy.SetLength(static_cast<uint16_t>(payload.length()));
181
182 uint16_t marker = NETX_HTONS(headerCopy.GetMarker());
183 uint16_t id = NETX_HTONS(headerCopy.GetId());
184 uint16_t seqNum = NETX_HTONS(headerCopy.GetSeqNum());
185 uint16_t length = NETX_HTONS(headerCopy.GetLength());
186
187 // 3. Append Data (With Error Checks)
188 // Check return value of every append. If one fails, the packet is corrupted/partial.
189 if (nx_packet_data_append(packet_ptr, &marker, sizeof(marker), m_pool, NX_NO_WAIT) != NX_SUCCESS)
190 goto cleanup_and_fail;
191
192 if (nx_packet_data_append(packet_ptr, &id, sizeof(id), m_pool, NX_NO_WAIT) != NX_SUCCESS)
193 goto cleanup_and_fail;
194
195 if (nx_packet_data_append(packet_ptr, &seqNum, sizeof(seqNum), m_pool, NX_NO_WAIT) != NX_SUCCESS)
196 goto cleanup_and_fail;
197
198 if (nx_packet_data_append(packet_ptr, &length, sizeof(length), m_pool, NX_NO_WAIT) != NX_SUCCESS)
199 goto cleanup_and_fail;
200
201 if (nx_packet_data_append(packet_ptr, (VOID*)payload.data(), payload.size(), m_pool, NX_NO_WAIT) != NX_SUCCESS)
202 goto cleanup_and_fail;
203
204 // 4. Send
205 // nx_udp_socket_send typically releases the packet on internal errors.
206 // However, if it fails due to binding issues (before driver), it might not.
207 ret = nx_udp_socket_send(&m_socket, packet_ptr, m_targetIp, m_port);
208
209 if (ret != NX_SUCCESS) {
210 // Defensive release: Verify if your specific NetX version auto-releases on specific errors.
211 // Standard practice is often to release if send fails.
212 nx_packet_release(packet_ptr);
213 tx_mutex_put(&m_mutex);
214 return -1;
215 }
216
217 // 5. Track Reliability (after successful send only)
218 if (headerCopy.GetId() != dmq::ACK_REMOTE_ID && m_transportMonitor)
219 m_transportMonitor->Add(headerCopy.GetSeqNum(), headerCopy.GetId());
220
221 tx_mutex_put(&m_mutex);
222 return 0;
223
224 cleanup_and_fail:
225 nx_packet_release(packet_ptr);
226 tx_mutex_put(&m_mutex);
227 return -1;
228 }

230 virtual int Receive(xstringstream& is, DmqHeader& header) override
231 {
232 tx_mutex_get(&m_mutex, TX_WAIT_FOREVER);
233
234 if (m_recvTransport != this) {
235 tx_mutex_put(&m_mutex);
236 return -1;
237 }
238
239 NX_PACKET* packet_ptr = nullptr;
240
241 // Use configured timeout
242 UINT ret = nx_udp_socket_receive(&m_socket, &packet_ptr, m_recvTimeout);
243
244 if (ret != NX_SUCCESS) {
245 tx_mutex_put(&m_mutex);
246 return -1; // No packet or error
247 }
248
249 // --- Extract Source Info for ACKs ---
250 ULONG srcIp;
251 UINT srcPort;
252 nx_udp_source_extract(packet_ptr, &srcIp, &srcPort);
253
254 if (m_type == Type::SUB) {
255 m_targetIp = srcIp;
256 m_port = srcPort;
257 }
258
259 // --- Read Data ---
260 ULONG bytesRead;
261 UCHAR buffer[1500];
262
263 // Validate total length before attempting retrieve
264 if (packet_ptr->nx_packet_length > sizeof(buffer)) {
265 nx_packet_release(packet_ptr);
266 tx_mutex_put(&m_mutex);
267 return -1; // Packet too large for buffer
268 }
269
270 // Retrieve copies data out of packet (handling chained packets if necessary)
271 ret = nx_packet_data_retrieve(packet_ptr, buffer, &bytesRead);
272
273 nx_packet_release(packet_ptr);
274
275 if (ret != NX_SUCCESS || bytesRead < DmqHeader::HEADER_SIZE) {
276 tx_mutex_put(&m_mutex);
277 return -1;
278 }
279
280 // Parse Header
281 uint16_t* ptr16 = (uint16_t*)buffer;
282 header.SetMarker(NETX_NTOHS(ptr16[0]));
283 header.SetId(NETX_NTOHS(ptr16[1]));
284 header.SetSeqNum(NETX_NTOHS(ptr16[2]));
285 header.SetLength(NETX_NTOHS(ptr16[3]));
286
287 if (header.GetMarker() != DmqHeader::MARKER) {
288 tx_mutex_put(&m_mutex);
289 return -1;
290 }
291
292 // Check payload bounds
293 if (bytesRead < (ULONG)(DmqHeader::HEADER_SIZE + header.GetLength())) {
294 tx_mutex_put(&m_mutex);
295 return -1;
296 }
297
298 // Reset stream state before writing
299 is.clear();
300 is.str("");
301
302 // Write Payload to stream
303 is.write((char*)(buffer + DmqHeader::HEADER_SIZE), header.GetLength());
304
305 // --- ACK Logic ---
306 if (header.GetId() == dmq::ACK_REMOTE_ID) {
307 if (m_transportMonitor) m_transportMonitor->Remove(header.GetSeqNum());
308 }
309 else if (m_transportMonitor && m_sendTransport) {
310 // Send ACK
311 xostringstream ss_ack;
312 DmqHeader ack;
313 ack.SetId(dmq::ACK_REMOTE_ID);
314 ack.SetSeqNum(header.GetSeqNum());
315 ack.SetLength(0);
316
317 // Release lock before recursive call to Send to prevent deadlock
318 tx_mutex_put(&m_mutex);
319 m_sendTransport->Send(ss_ack, ack);
320 return 0;
321 }
322
323 tx_mutex_put(&m_mutex);
324 return 0;
325 }
