110 bool Start(const char* nodeName,
111 uint16_t listenPort,
112 dmq::Duration watchdog = std::chrono::seconds(30),
113 dmq::Duration tickPeriod = std::chrono::milliseconds(10))
114 {
115 dmq::LockGuard<dmq::RecursiveMutex> lock(m_mutex);
116 if (m_running) return true;
117
118 char threadName[64];
119 snprintf(threadName, sizeof(threadName), "%s_NetworkThread", nodeName);
120
121 if (m_recvTransport.Create(Transport::Type::SUB, "127.0.0.1", listenPort) != 0) {
122 printf("NetworkNode [%s]: ERROR - failed to open recv port %u\n", nodeName, listenPort);
123 return false;
124 }
125 m_recvTransport.SetRecvTimeout(std::chrono::milliseconds(1));
126 // Allow the SUB socket to send ACKs back to senders of RELIABLE messages.
127 // The `else if (m_sendTransport)` gate in Receive() requires a non-null
128 // m_sendTransport; using self means ACKs go out via sendto on this socket.
129 m_recvTransport.SetSendTransport(&m_recvTransport);
130
131 m_recvParticipant = dmq::xmake_shared<Participant>(m_recvTransport);
132
133 // Apply incoming topics registered before Start().
134 for (size_t i = 0; i < m_inCount; ++i)
135 m_inTopics[i].adder(m_inTopics[i].serializer,
136 m_inTopics[i].topic,
137 m_inTopics[i].remoteId,
138 *m_recvParticipant);
139
140 m_thread.emplace(
141 threadName, 100, dmq::os::FullPolicy::FAULT, dmq::DEFAULT_DISPATCH_TIMEOUT);
142
143 m_running = true;
144 m_tickPeriod = tickPeriod;
145
146 std::optional<dmq::Duration> wd = (watchdog.count() > 0)
147 ? std::optional<dmq::Duration>{watchdog}
148 : std::nullopt;
149 m_thread->CreateThread(wd);
150
151 m_recvConn = m_recvTimer.OnExpired.Connect(
152 dmq::util::MakeTimerDelegate(this, &NetworkNode::ReceiverThread, *m_thread));
153 m_recvTimer.Start(tickPeriod);
154
155 return true;
156 }

159 void Stop()
160 {
161 {
162 dmq::LockGuard<dmq::RecursiveMutex> lock(m_mutex);
163 if (!m_running) return;
164 m_running = false;
165
166 m_recvTimer.Stop();
167 m_recvConn.Disconnect();
168 m_recvTransport.Close();
169 }
170
171 // Exit the receive thread with no lock held. ReceiverThread() is dispatched
172 // asynchronously via TimerDelegate, so a tick may already be queued on
173 // m_thread when Stop() is called; ExitThread() blocks (join()) until the
174 // thread's dispatch loop drains its queue, which includes running any
175 // already-queued ReceiverThread() tick to completion. That tick acquires
176 // m_mutex too, so holding it here while blocked in join() would deadlock.
177 if (m_thread)
178 m_thread->ExitThread();
179
180 dmq::LockGuard<dmq::RecursiveMutex> lock(m_mutex);
181 for (size_t i = m_peerCount; i > 0; --i) {
182 RemoteNode& node = m_peers[i - 1];
183 if (!node.active) continue;
184 DataBus::RemoveParticipant(node.reliableParticipant);
185 DataBus::RemoveParticipant(node.unreliableParticipant);
186 if (node.reliableParticipant) node.reliableParticipant->SetSendThread(nullptr);
187 if (node.unreliableParticipant) node.unreliableParticipant->SetSendThread(nullptr);
188 node.capConn.Disconnect();
189 node.pendingConn.Disconnect();
190 node.rawTransport.Close();
191 node.active = false;
192 }
193 m_peerCount = 0;
194 }

204 void AddPeer(const char* name, const char* addr, uint16_t port)
205 {
206 dmq::LockGuard<dmq::RecursiveMutex> lock(m_mutex);
207 ASSERT_TRUE(m_peerCount < MaxPeers);
208
209 RemoteNode& node = m_peers[m_peerCount];
210 strncpy(node.name, name, sizeof(node.name) - 1);
211 node.name[sizeof(node.name) - 1] = '\0';
212
213 if (node.rawTransport.Create(Transport::Type::PUB, addr, port) != 0) {
214 printf("NetworkNode: ERROR - failed to connect to peer '%s' at %s:%u\n", name, addr, port);
215 return;
216 }
217
218 // Wire the transport monitor so Send() calls TransportMonitor::Add(seqNum)
219 // and incoming ACKs call TransportMonitor::Remove(seqNum). Without this,
220 // m_pending stays empty, Process() never fires timeouts, and RetryMonitor
221 // entries are never erased.
222 node.rawTransport.SetTransportMonitor(&node.transportMonitor);
223
224 // Allow Receive() to be called on the PUB socket so ReceiverThread can drain
225 // ACKs. Remote peers send ACKs to our PUB socket's OS-assigned ephemeral port
226 // (the source address they saw when the message arrived), not to our SUB listen
227 // port. Without this, Remove() is never called and m_pending fills to cap.
228 node.rawTransport.SetRecvTransport(&node.rawTransport);
229
230 // Two-phase init: wire the reliability stack using stable member addresses.
231 // RemoteNode is stored by value in std::array — elements never relocate.
232 node.retryMonitor.Init(node.rawTransport, node.transportMonitor);
233 node.reliableTransport.Init(node.rawTransport, node.retryMonitor);
234
235 dmq::xstring peerName(name);
236 node.capConn = node.transportMonitor.OnCapExceeded.Connect(
237 dmq::MakeDelegate([peerName](size_t n) {
238 printf("NetworkNode [%s]: cap exceeded (%zu unacked messages)\n",
239 peerName.c_str(), n);
240 }));
241 node.pendingConn = node.transportMonitor.OnPendingExceeded.Connect(
242 dmq::MakeDelegate([peerName](size_t n) {
243 printf("NetworkNode [%s]: pending exceeded (%zu entries remain)\n",
244 peerName.c_str(), n);
245 }));
246
247 node.reliableParticipant = dmq::xmake_shared<Participant>(node.reliableTransport);
248 node.unreliableParticipant = dmq::xmake_shared<Participant>(node.rawTransport);
249
250 if (m_thread.has_value()) {
251 node.reliableParticipant->SetSendThread(&*m_thread);
252 node.unreliableParticipant->SetSendThread(&*m_thread);
253 }
254
255 DataBus::AddParticipant(node.reliableParticipant);
256 DataBus::AddParticipant(node.unreliableParticipant);
257
258 // Apply outgoing topics registered before this peer was added.
259 for (size_t i = 0; i < m_outCount; ++i) {
260 const OutgoingTopic& out = m_outTopics[i];
261 if (out.reliability == Reliability::RELIABLE)
262 node.reliableParticipant->AddRemoteTopic(out.topic, out.remoteId);
263 else
264 node.unreliableParticipant->AddRemoteTopic(out.topic, out.remoteId);
265 }
266
267 node.active = true;
268 m_peerCount++;
269 }

280 template <typename T>
281 void Send(const dmq::xstring& topic,
282 dmq::DelegateRemoteId remoteId,
283 dmq::ISerializer<void(T)>& serializer,
284 Reliability rel = Reliability::UNRELIABLE)
285 {
286 dmq::LockGuard<dmq::RecursiveMutex> lock(m_mutex);
287 ASSERT_TRUE(m_outCount < MaxTopics);
288 DataBus::RegisterSerializer<T>(topic, serializer);
289 m_outTopics[m_outCount++] = {topic, remoteId, rel};
290
291 // Apply to peers already added.
292 for (size_t i = 0; i < m_peerCount; ++i) {
293 RemoteNode& node = m_peers[i];
294 if (!node.active) continue;
295 if (rel == Reliability::RELIABLE)
296 node.reliableParticipant->AddRemoteTopic(topic, remoteId);
297 else
298 node.unreliableParticipant->AddRemoteTopic(topic, remoteId);
299 }
300 }

310 template <typename T>
311 void Receive(const dmq::xstring& topic,
312 dmq::DelegateRemoteId remoteId,
313 dmq::ISerializer<void(T)>& serializer)
314 {
315 dmq::LockGuard<dmq::RecursiveMutex> lock(m_mutex);
316 ASSERT_TRUE(m_inCount < MaxTopics);
317
318 // Store a type-erased adder so it can be replayed when Start() creates the participant.
319 m_inTopics[m_inCount++] = {
320 topic, remoteId, static_cast<void*>(&serializer),
321 [](void* ser, const dmq::xstring& t, dmq::DelegateRemoteId rid, Participant& p) {
322 DataBus::AddIncomingTopic<T>(t, rid, p,
323 *static_cast<dmq::ISerializer<void(T)>*>(ser));
324 }
325 };
326
327 // Apply immediately if Start() has already run.
328 if (m_recvParticipant)
329 DataBus::AddIncomingTopic<T>(topic, remoteId, *m_recvParticipant, serializer);
330 }

332private:
333 void ReceiverThread()
334 {
335 dmq::LockGuard<dmq::RecursiveMutex> lock(m_mutex);
336 // Override via DMQ_NETWORK_NODE_MAX_WORK in delegatemqconfig.h.
337 constexpr int MAX_WORK = dmq::NETWORK_NODE_MAX_WORK;
338
339 if (m_recvParticipant) {
340 for (int i = 0; i < MAX_WORK; ++i)
341 if (m_recvParticipant->ProcessIncoming() != 0) break;
342 }
343
344 // Drain ACKs that arrived on each peer's PUB socket (ephemeral source port).
345 // Remote peers send ACKs to the source address they received from, which is
346 // our PUB socket's OS-assigned port — not our SUB listen port. Receive() on
347 // the PUB socket calls TransportMonitor::Remove(seqNum) when an ACK arrives,
348 // firing SUCCESS so RetryMonitor erases the entry. The 2 ms recv timeout on
349 // PUB sockets keeps each poll fast when there is nothing to receive.
350 for (size_t i = 0; i < m_peerCount; ++i) {
351 if (!m_peers[i].active) continue;
352 for (int k = 0; k < MAX_WORK; ++k) {
353 dmq::transport::DmqHeader ackHeader;
354 dmq::xstringstream ackStream(std::ios::in | std::ios::out | std::ios::binary);
355 if (m_peers[i].rawTransport.Receive(ackStream, ackHeader) != 0) break;
356 }
357 }
358
359 // Throttled: run TransportMonitor::Process() ~10 times per second.
360 auto now = dmq::Clock::now();
361 if (now - m_lastMonitorCheck >= std::chrono::milliseconds(100)) {
362 for (size_t i = 0; i < m_peerCount; ++i) {
363 if (m_peers[i].active)
364 m_peers[i].transportMonitor.Process();
365 }
366 m_lastMonitorCheck = now;
367 }
368 }

370 // -----------------------------------------------------------------------
371 // RemoteNode — fully by-value storage; no heap for transport objects.
372 //
373 // std::array guarantees elements are never relocated after construction,
374 // so Init() references between sibling members remain valid for the
375 // lifetime of this NetworkNode:
376 // retryMonitor refs rawTransport + transportMonitor
377 // reliableTransport refs rawTransport + retryMonitor
378 // -----------------------------------------------------------------------
379 struct RemoteNode {
380 Transport rawTransport;
381 TransportMonitor transportMonitor;
382 RetryMonitor retryMonitor; // Init(rawTransport, transportMonitor)
383 ReliableTransport reliableTransport; // Init(rawTransport, retryMonitor)
384 dmq::ScopedConnection capConn;

387 std::shared_ptr<Participant> unreliableParticipant; // xmake_shared on construction
388 bool active = false;
389 char name[48] = {};
390 };

395
396 struct IncomingTopic {
397 dmq::xstring topic;
398 dmq::DelegateRemoteId remoteId = 0;

401 };

403 struct OutgoingTopic {
404 dmq::xstring topic;
405 dmq::DelegateRemoteId remoteId = 0;
406 Reliability reliability = Reliability::UNRELIABLE;
407 };
408
409 // Inbound path — one participant handles all incoming traffic on listenPort.
410 Transport m_recvTransport;
411 std::shared_ptr<Participant> m_recvParticipant;
412
413 // Peer table — fixed-size; no heap for transport or reliability-stack objects.
414 std::array<RemoteNode, MaxPeers> m_peers{};

416
417 // Topic tables — fixed-size.
418 std::array<IncomingTopic, MaxTopics> m_inTopics{};

421 size_t m_outCount = 0;
422
423 // Receive thread. Delay construction until Start() via std::optional to
424 // avoid heap allocation while allowing runtime thread naming.
425 std::optional<dmq::os::Thread> m_thread;
426 dmq::util::Timer m_recvTimer;
427 dmq::ScopedConnection m_recvConn;
428 dmq::Duration m_tickPeriod{};

431 mutable dmq::RecursiveMutex m_mutex;
432};
