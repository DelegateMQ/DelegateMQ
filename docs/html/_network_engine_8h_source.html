214 template <class Target, class... Args>
215 bool RemoteInvokeWaitInternal(Target& target, Args&&... args)
216 {
217 // [Network Thread] Already on the correct thread: send immediately,
218 // no blocking wait is possible (we would deadlock waiting on ourselves).
219 if (m_thread.IsCurrentThread())
220 {
221 target(std::forward<Args>(args)...);
222 return (target.GetError() == dmq::DelegateError::SUCCESS);
223 }
224
225 // 1. [Caller Thread] Create shared synchronization state.
226 auto state = dmq::xmake_shared<InvokeWaitState>();
227 dmq::DelegateRemoteId remoteId = target.GetRemoteId();
228
229 // 2. [Caller Thread] Define the status callback that wakes us up later.
230 // Fires on the receive thread (ACK) or timer thread (timeout).
231 std::function<void(dmq::DelegateRemoteId, uint16_t, TransportMonitor::Status)> statusCbFunc =
232 [state, remoteId](dmq::DelegateRemoteId id, uint16_t seq, TransportMonitor::Status status) {
233 if (id != remoteId)
234 return;
235 bool notify = false;
236 {
237 dmq::LockGuard<dmq::Mutex> lock(state->mtx);
238 if (!state->seqSet) {
239 // Send thread has not recorded the seq yet; buffer the
240 // status so the send lambda can reconcile it.
241 if (state->earlyCnt < state->early.size())
242 state->early[state->earlyCnt++] = { seq, status };
243 }
244 else if (!state->complete && seq == state->expectedSeq) {
245 state->complete = true;
246 state->success = (status == TransportMonitor::Status::SUCCESS);
247 notify = true;
248 }
249 }
250 if (notify)
251 state->cv.notify_one();
252 };
253
254 // 3. [Caller Thread] Register the callback.
255 dmq::ScopedConnection conn = m_transportMonitor.OnSendStatus.Connect(dmq::MakeDelegate(statusCbFunc));
256
257 // 4. [Caller Thread] Define the "Send" logic lambda.
258 auto* targetPtr = &target;
259 std::function<bool(Args...)> asyncCallFunc = [targetPtr, state](auto&&... fwdArgs) -> bool {
260 (*targetPtr)(std::forward<decltype(fwdArgs)>(fwdArgs)...);
261 bool notify = false;
262 {
263 dmq::LockGuard<dmq::Mutex> lock(state->mtx);
264 state->expectedSeq = targetPtr->GetLastSeqNum();
265 state->seqSet = true;
266 // Reconcile any status that arrived before the seq was known
267 for (size_t i = 0; i < state->earlyCnt; i++) {
268 if (state->early[i].first == state->expectedSeq) {
269 state->complete = true;
270 state->success = (state->early[i].second == TransportMonitor::Status::SUCCESS);
271 notify = true;
272 break;
273 }
274 }
275 }
276 if (notify)
277 state->cv.notify_one();
278 return (targetPtr->GetError() == dmq::DelegateError::SUCCESS);
279 };
280
281 // 5. [Caller Thread] Dispatch the lambda to the Network Thread queue.
282 auto retVal = dmq::MakeDelegate(asyncCallFunc, m_thread, SEND_TIMEOUT)
283 .AsyncInvoke(std::forward<Args>(args)...);
284
285 if (retVal.has_value() && retVal.value() == true)
286 {
287 // 6. [Caller Thread] BLOCK until the status callback (or the send
288 // lambda's reconciliation) completes the wait, or timeout.
289 dmq::UniqueLock<dmq::Mutex> lock(state->mtx);
290 state->cv.wait_for(lock, RECV_TIMEOUT, [&] { return state->complete; });
291 return state->success;
292 }
293
294 // 7. [Caller Thread] Send failed or timed out queueing to the network
295 // thread. 'conn' disconnects on scope exit; 'state' stays alive via
296 // shared_ptr if a late status callback races with our return.
297 dmq::LockGuard<dmq::Mutex> lock(state->mtx);
298 return state->success;
299 }
