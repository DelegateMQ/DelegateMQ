149 template <class TClass, class RetType, class... Args>
150 bool RemoteInvokeWait(dmq::DelegateMemberRemote<TClass, RetType(Args...)>& endpoint, Args&&... args)
151 {
152 // 1. [Caller Thread] Check if we are on the Network Thread.
153 if (!m_thread.IsCurrentThread())
154 {
155 // 2. [Caller Thread] Create shared synchronization state.
156 struct SyncState {
157 std::atomic<bool> success{ false };
158 bool complete = false;
159 dmq::Mutex mtx; // Generic Mutex
160 dmq::ConditionVariable cv; // Generic CV
161 XALLOCATOR
162 };
163 auto state = dmq::xmake_shared<SyncState>();
164 dmq::DelegateRemoteId remoteId = endpoint.GetRemoteId();
165
166 // 3. [Caller Thread] Define the callback that wakes us up later.
167 std::function<void(dmq::DelegateRemoteId, uint16_t, TransportMonitor::Status)> statusCbFunc =
168 [state, remoteId](dmq::DelegateRemoteId id, uint16_t seq, TransportMonitor::Status status) {
169 if (id == remoteId) {
170 {
171 dmq::LockGuard<dmq::Mutex> lock(state->mtx);
172 state->complete = true;
173 if (status == TransportMonitor::Status::SUCCESS)
174 state->success.store(true);
175 }
176 // 9. [Network Thread] Notify the waiting caller thread.
177 state->cv.notify_one();
178 }
179 };
180
181 // 4. [Caller Thread] Register the callback.
182 dmq::ScopedConnection conn = m_transportMonitor.OnSendStatus.Connect(dmq::MakeDelegate(statusCbFunc));
183
184 // 5. [Caller Thread] Define the "Send" logic lambda.
185 auto* epPtr = &endpoint;
186 std::function<bool(Args...)> asyncCallFunc = [epPtr](auto&&... fwdArgs) -> bool {
187 // 7. [Network Thread] Execute the send operation.
188 (*epPtr)(std::forward<decltype(fwdArgs)>(fwdArgs)...);
189 return (epPtr->GetError() == dmq::DelegateError::SUCCESS);
190 };
191
192 // 6. [Caller Thread] Dispatch the lambda to the Network Thread queue.
193 auto retVal = dmq::MakeDelegate(asyncCallFunc, m_thread, SEND_TIMEOUT)
194 .AsyncInvoke(std::forward<Args>(args)...);
195
196 if (retVal.has_value() && retVal.value() == true)
197 {
198 // 8. [Caller Thread] BLOCK and Wait.
199 dmq::UniqueLock<dmq::Mutex> lock(state->mtx);
200 // wait_for returns false if the predicate is still false after the timeout
201 state->cv.wait_for(lock, RECV_TIMEOUT, [&] {
202 return state->complete;
203 });
204 // 10. [Caller Thread] Wake up! The wait is over.
205 }
206
207 // 11. [Caller Thread] Cleanup and return result.
208 // 'conn' goes out of scope here and automatically disconnects
209 return state->success.load();
210 }
211 else
212 {
213 // 12. [Network Thread] Alternative Path:
214 // We are already on the correct thread, so execute immediately.
215 endpoint(std::forward<Args>(args)...);
216 return (endpoint.GetError() == dmq::DelegateError::SUCCESS);
217 }
218 }

254 auto* chPtr = &channel;

256 (*chPtr)(std::forward<decltype(fwdArgs)>(fwdArgs)...);
257 return (chPtr->GetError() == dmq::DelegateError::SUCCESS);
258 };

264 {
265 dmq::UniqueLock<dmq::Mutex> lock(state->mtx);
266 state->cv.wait_for(lock, RECV_TIMEOUT, [&] { return state->complete; });
