107 void operator()(Args... args) {
108 // To prevent deadlocks, the mutex must be released before invoking the
109 // delegate. Circular lock dependencies can occur if the delegate target
110 // function itself attempts to acquire a lock that is held by the
111 // thread invoking the delegate.
112 // Use a small-buffer optimization to avoid heap allocation in the common case.
113 // Buffers hold `DelegateBase` (not `DelegateType`) so this `xlist` instantiation
114 // is shared across every `MulticastDelegateSafe<Sig>` signature; only the final
115 // invoke below needs the concrete `DelegateType` back. The snapshot-under-lock
116 // logic itself lives in `detail::MulticastSafeSnapshot` for the same reason.
117 std::shared_ptr<DelegateBase> small_buf[SIGNAL_SBO_COUNT];
118 xlist<std::shared_ptr<DelegateBase>> large_buf;
119 size_t count = detail::MulticastSafeSnapshot(this->m_delegates, m_lock, small_buf, SIGNAL_SBO_COUNT, large_buf);
120
121 if (count <= SIGNAL_SBO_COUNT) {
122 for (size_t i = 0; i < count; ++i) {
123 if (small_buf[i])
124 (*static_cast<DelegateType*>(small_buf[i].get()))(args...);
125 small_buf[i].reset(); // Clear to release shared_ptr immediately
126 }
127 } else {
128 for (auto& d : large_buf) {
129 if (d)
130 (*static_cast<DelegateType*>(d.get()))(args...);
131 }
132 }
133 }
