341 m_retVal.reset();
342 m_success = false;
343 if (this->Empty())
344 return RetType();
345
346 // Synchronously invoke the target function?
347 if (m_sync) {
348 // Invoke the target function directly
349 return BaseType::operator()(std::forward<Args>(args)...);
350 } else {
351 // Create a clone instance of this delegate
352 auto delegate = xmake_shared<ClassType>(*this);
353 if (!delegate)
354 BAD_ALLOC();
355
356 // Create a new message instance for sending to the destination thread.
357 auto msg = xmake_shared<DelegateAsyncWaitMsg<Args...>>(delegate, m_priority, std::forward<Args>(args)...);
358 if (!msg)
359 BAD_ALLOC();
360 msg->SetInvokerWaiting(true);
361
362 bool waited = false;
363 auto thread = this->GetThread();
364 if (thread) {
365 // Dispatch message onto the callback destination thread. Invoke()
366 // will be called by the destination thread.
367 if (thread->DispatchDelegate(msg)) {
368 // Wait for destination thread to execute the delegate function and get return value
369 waited = msg->GetSema().Wait(m_timeout);
370 }
371 }
372
373 // Single lock: read return value and clear InvokerWaiting atomically
374 const dmq::LockGuard<Mutex> lock(msg->GetLock());
375 if (waited) {
376 // Only report success if the target function invoke actually completed;
377 // the semaphore is signaled even when the target function threw, so
378 // `waited` alone is not sufficient to know the call succeeded.
379 m_success = msg->GetInvokeSucceeded();
380 m_retVal = delegate->m_retVal;
381 }
382 // Set flag that source is not waiting anymore
383 msg->SetInvokerWaiting(false);
384
385 // Does the target function have a return value?
386 if constexpr (std::is_void<RetType>::value == false) {
387 // Is the return value valid?
388 if (m_retVal.has_value()) {
389 // Return the destination thread target function return value
390 return GetRetVal();
391 } else {
392 // Return a default return value

394 }

426 virtual bool Invoke(std::shared_ptr<DelegateMsg> msg) override {
427 static_assert(!(is_unique_ptr<RetType>::value), "std::unique_ptr return value not allowed");
428
429 // Typecast the base pointer to back correct derived to instance
430 auto delegateMsg = std::dynamic_pointer_cast<DelegateAsyncWaitMsg<Args...>>(msg);
431 if (delegateMsg == nullptr)
432 return false;
433
434 // Protect data shared between source and destination threads
435 const dmq::LockGuard<Mutex> lock(delegateMsg->GetLock());
436
437 // Is the source thread waiting for the target function invoke to complete?
438 if (delegateMsg->GetInvokerWaiting()) {
439 // Invoke the delegate function synchronously
440 m_sync = true;
441
442 // Signals the source thread when this scope exits, including via an exception
443 // propagating out of the target function invoke below. Without this, a target
444 // function that throws would leave the source thread blocked until timeout.
445 struct SemaSignalGuard {
446 Semaphore& sema;
447 ~SemaSignalGuard() { sema.Signal(); }
448 } semaSignalGuard{ delegateMsg->GetSema() };
449
450 // Does target function have a void return value?
451 if constexpr (std::is_void<RetType>::value == true) {
452 // Invoke the target function using the source thread supplied function arguments
453 std::apply(&BaseType::operator(), std::tuple_cat(std::make_tuple(this), delegateMsg->GetArgs()));
454 } else {
455 // Invoke the target function using the source thread supplied function arguments
456 // and get the return value
457 m_retVal = std::apply(&BaseType::operator(), std::tuple_cat(std::make_tuple(this), delegateMsg->GetArgs()));
458 }
459
460 // Only reached if the std::apply() call above did not throw
461 delegateMsg->SetInvokeSucceeded(true);
462 }
