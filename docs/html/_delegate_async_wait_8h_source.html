226template <class... Args>
227bool DispatchAsyncWait(std::shared_ptr<IThreadInvoker> invoker, IThread* thread, Priority priority,
228 Duration timeout, Args&&... args) {
229 // Create a new message instance for sending to the destination thread. Erase to
230 // the non-templated DelegateMsg base BEFORE constructing the shared_ptr (rather
231 // than xmake_shared<DelegateAsyncWaitMsg<Args...>>) so the shared_ptr control
232 // block is not templated on Args... -- one control-block type is then shared
233 // across every DelegateAsyncWait signature instead of duplicated per signature.
234 // `rawMsg` stays valid for the whole function (kept alive by `msg`) and is used
235 // for every access specific to the concrete DelegateAsyncWaitMsg<Args...> type;
236 // `msg` itself is only needed, as the erased base, for DispatchDelegate().
237 auto* rawMsg = new(std::nothrow) DelegateAsyncWaitMsg<Args...>(std::move(invoker), priority, std::forward<Args>(args)...);
238 if (!rawMsg)
239 BAD_ALLOC();
240 DelegateMsg* msgBase = rawMsg;
241 std::shared_ptr<DelegateMsg> msg(msgBase, std::default_delete<DelegateMsg>(), dmq::stl_allocator<DelegateMsg>());
242 rawMsg->SetInvokerWaiting(true);
243
244 bool waited = false;
245 if (thread) {
246 // Dispatch message onto the callback destination thread. Invoke()
247 // will be called by the destination thread.
248 if (thread->DispatchDelegate(msg)) {
249 // Wait for destination thread to execute the delegate function and get return value
250 waited = rawMsg->GetSema().Wait(timeout);
251 }
252 }
253
254 // Single lock: read return value and clear InvokerWaiting atomically
255 const dmq::LockGuard<Mutex> lock(rawMsg->GetLock());
256 // Only report success if the target function invoke actually completed;
257 // the semaphore is signaled even when the target function threw, so
258 // `waited` alone is not sufficient to know the call succeeded.
259 bool success = waited && rawMsg->GetInvokeSucceeded();
260 // Set flag that source is not waiting anymore
261 rawMsg->SetInvokerWaiting(false);
262 return success;
263}
