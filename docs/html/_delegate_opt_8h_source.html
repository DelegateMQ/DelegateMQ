208 // --- PORTABLE LOCK GUARD ---
209 // Does not require <mutex>. Works with any BasicLockable type (lock/unlock).
210 template<typename T>
211 class PortableLockGuard {
212 T& m_mutex;
213 public:
214 explicit PortableLockGuard(T& m) noexcept : m_mutex(m) { m_mutex.lock(); }
215 ~PortableLockGuard() noexcept { m_mutex.unlock(); }
216 PortableLockGuard(const PortableLockGuard&) = delete;
217 PortableLockGuard& operator=(const PortableLockGuard&) = delete;
218 };

220 // --- PORTABLE SCOPED LOCK ---
221 // Maps to std::scoped_lock on all known threaded ports (desktop + RTOS).
222 // All RTOS mutex types satisfy BasicLockable (lock/unlock), so std::scoped_lock
223 // works with them directly. Falls back to a no-op for DMQ_THREAD_NONE and for
224 // any unrecognized bare-metal build where no thread model is defined.
225#if defined(DMQ_HAS_THREADING)
226 template <typename... M>
227 using ScopedLock = std::scoped_lock<M...>;
228#else
229 // No-op: DMQ_THREAD_NONE or no thread model defined (bare-metal, single-threaded)
230 template <typename... M>
231 class ScopedLock {
232 public:
233 explicit ScopedLock(M&...) noexcept {}
234 ~ScopedLock() noexcept = default;
235 ScopedLock(const ScopedLock&) = delete;
236 ScopedLock& operator=(const ScopedLock&) = delete;
237 };

272 // --- THIS_THREAD (sleep_for / yield) SELECTION ---
273 // Portable equivalents of std::this_thread::sleep_for()/yield() for the
274 // calling thread. Exists so library internals that need to delay or
275 // yield (e.g. RetryMonitor backoff) aren't forced to pull in a full
276 // dmq::os::Thread -- which also drags in the message queue, watchdog,
277 // and stats machinery -- just to sleep. Each dmq::os::Thread::Sleep()
278 // forwards here too, so every port's native delay call is implemented
279 // exactly once.
280 //
281 // A class (not a "this_thread" namespace) deliberately: sample/app code
282 // throughout this project does `using namespace dmq; ... using namespace
283 // std;` together, and a same-named nested namespace would make unqualified
284 // this_thread::sleep_for() calls in that code ambiguous against
285 // std::this_thread.
286#if defined(DMQ_THREAD_STDLIB) || defined(DMQ_THREAD_WIN32) || defined(DMQ_THREAD_QT)
287 // Windows / Linux / macOS / Qt -- std::this_thread is already portable here.
288 struct ThisThread {
289 template<typename Rep, typename Period>
290 static void sleep_for(std::chrono::duration<Rep, Period> d) { std::this_thread::sleep_for(d); }
291 static void yield() noexcept { std::this_thread::yield(); }
292 };
293
294#elif defined(DMQ_THREAD_FREERTOS)
295 struct ThisThread {
296 template<typename Rep, typename Period>
297 static void sleep_for(std::chrono::duration<Rep, Period> d) {
298 dmq::os::FreeRTOSThisThread::sleep_for(std::chrono::duration_cast<std::chrono::milliseconds>(d));
299 }
300 static void yield() noexcept { dmq::os::FreeRTOSThisThread::yield(); }
301 };
302
303#elif defined(DMQ_THREAD_THREADX)
304 struct ThisThread {
305 template<typename Rep, typename Period>
306 static void sleep_for(std::chrono::duration<Rep, Period> d) {
307 dmq::os::ThreadXThisThread::sleep_for(std::chrono::duration_cast<std::chrono::milliseconds>(d));
308 }
309 static void yield() noexcept { dmq::os::ThreadXThisThread::yield(); }
310 };
311
312#elif defined(DMQ_THREAD_ZEPHYR)
313 struct ThisThread {
314 template<typename Rep, typename Period>
315 static void sleep_for(std::chrono::duration<Rep, Period> d) {
316 dmq::os::ZephyrThisThread::sleep_for(std::chrono::duration_cast<std::chrono::milliseconds>(d));
317 }
318 static void yield() noexcept { dmq::os::ZephyrThisThread::yield(); }
319 };
320
321#elif defined(DMQ_THREAD_CMSIS_RTOS2)

324 static void sleep_for(std::chrono::duration<Rep, Period> d) {
