29 // Set a thread to serialize all outgoing sends.
30 // When set, Send() posts the channel invocation to this thread rather than
31 // executing inline on the caller's thread. This provides two benefits:
32 // 1. Thread safety: concurrent Publish() calls from multiple threads are
33 // safe — each send is queued rather than racing on the shared stream.
34 // 2. Non-blocking callers: blocking I/O in the transport cannot stall
35 // unrelated publishing threads.
36 // Without a send thread, concurrent calls to Send() for the same remote ID
37 // are a data race on the channel's internal stream buffer.
38 void SetSendThread(dmq::IThread* thread) { m_sendThread = thread; }

59 // Process incoming data from the transport.
60 // Must not be called concurrently — m_receiveMutex serializes the transport read.
61 // @return The result code from ITransport::Receive.
62 int ProcessIncoming() {
63 dmq::transport::DmqHeader header;
64 dmq::xstringstream inputStream(std::ios::in | std::ios::out | std::ios::binary);
65
66 int result;
67 {
68 // Serialize access to the transport read
69 dmq::LockGuard<dmq::Mutex> receiveLock(m_receiveMutex);
70 result = m_transport->Receive(inputStream, header);
71 }
72
73 if (result == 0) {
74 // Validate header marker
75 if (header.GetMarker() != dmq::transport::DmqHeader::MARKER) {
76 return -1; // Protocol error
77 }
78
79 dmq::DelegateRemoteId id = header.GetId();
80 uint16_t seqNum = header.GetSeqNum();
81
82 // Filter out duplicate messages (retries)
83 if (id != dmq::ACK_REMOTE_ID) {
84 dmq::LockGuard<dmq::RecursiveMutex> lock(m_mutex);
85 if (m_history[id].is_duplicate(seqNum)) {
86 return 0; // Silently drop duplicate
87 }
88 }
89
90 dmq::IRemoteInvoker* invoker = nullptr;
91 std::shared_ptr<void> channelLifetime; // keeps channel alive across lock gap
92 {
93 dmq::LockGuard<dmq::RecursiveMutex> lock(m_mutex);
94 auto it = m_channels.find(id);
95 if (it != m_channels.end()) {
96 channelLifetime = it->second.channel;
97 invoker = it->second.invoker;
98 }
99 }
100
101 // Invoke outside the lock to prevent deadlocks and allow re-entry
102 if (invoker) {
103 invoker->Invoke(inputStream);
104 } else {
105 OnChannelError(id, dmq::DelegateError::ERR_NO_DISPATCHER, 0);
106 }
107 }
108 return result;
109 }

111 // Send data for a topic.
112 // If a send thread was set via SetSendThread(), the channel invocation is
113 // dispatched asynchronously to that thread. Otherwise it executes inline.
114 template <typename T>
115 void Send(const dmq::xstring& topic, const T& data, dmq::ISerializer<void(T)>& serializer) {
116 dmq::DelegateRemoteId remoteId;
117 if (GetRemoteId(topic, remoteId)) {
118 auto channel = GetOrCreateChannel<T>(remoteId, serializer);
119 if (channel) {
120 if (m_sendThread) {
121 // Use a member function to avoid potential shared_ptr circular
122 // references. Pass arguments to AsyncInvoke.
123 (void)dmq::MakeDelegate(this, &Participant::AsyncSend<T>, *m_sendThread).AsyncInvoke(channel, data);
124 } else {
125 (*channel)(data);
126 }
127 }
128 }
129 }

209 void OnChannelError(dmq::DelegateRemoteId id, dmq::DelegateError error, dmq::DelegateErrorAux) {
210 if (error == dmq::DelegateError::SUCCESS) return;
211 bool shouldFire = false;
212 dmq::xstring topic;
213 {
214 dmq::LockGuard<dmq::RecursiveMutex> lock(m_mutex);
215 for (auto& [t, rid] : m_topicToRemoteId) {
216 if (rid == id) { topic = t; break; }
217 }
218 if (topic.empty()) {
219 // xostringstream keeps this portable: under DMQ_ALLOCATOR, xstring and
220 // std::string are different types, so std::to_string concatenation
221 // would not compile.
222 dmq::xostringstream os;
223 os << "UnknownRemoteId:" << id;
224 topic = os.str();
225 }
226 uint16_t bit = uint16_t(1u << static_cast<int>(error));
227 auto& bits = m_reportedErrors[topic];
228 if (!(bits & bit)) {
229 bits |= bit;
230 shouldFire = true;
231 } else if (m_continuousErrors) {
232 shouldFire = true;
233 }
234 }
235 if (shouldFire)
236 m_errorSignal(topic, error);
237 }

249 template <typename T>
250 std::shared_ptr<dmq::RemoteChannel<void(T)>> GetOrCreateChannel(dmq::DelegateRemoteId remoteId, dmq::ISerializer<void(T)>& serializer) {
251 dmq::LockGuard<dmq::RecursiveMutex> lock(m_mutex);
252 std::shared_ptr<dmq::RemoteChannel<void(T)>> channel;
253
254 auto it = m_channels.find(remoteId);
255 if (it != m_channels.end()) {
256 // Type safety: catch remoteId reused with a different T
257 auto itType = m_channelTypes.find(remoteId);
258 if (itType != m_channelTypes.end()) {
259 if (itType->second != std::type_index(typeid(T))) {
260 // Report before faulting so error handlers can log the mismatch
261 OnChannelError(remoteId, dmq::DelegateError::ERR_TYPE_MISMATCH, 0);
262 ASSERT();
263 }
264 }
265 channel = std::static_pointer_cast<dmq::RemoteChannel<void(T)>>(it->second.channel);
266 } else {
267 channel = dmq::xmake_shared<dmq::RemoteChannel<void(T)>>(*m_transport, serializer);
268
269 // Establish the remote ID for sending via operator().
270 channel->SetRemoteId(remoteId);
271
272 m_channels[remoteId] = { channel, channel->GetEndpoint() };
273 m_channelTypes.emplace(remoteId, std::type_index(typeid(T)));
274
275 // Ensure error handler is attached to the new channel
276 AttachErrorHandler(channel);
277 }
278
279 return channel;
280 }

293 // --- Duplicate Filtering ---
294 struct SeqHistory {
295 static constexpr size_t SIZE = DMQ_SEQ_HISTORY_SIZE;
296 uint16_t buffer[SIZE];
297 bool valid[SIZE];
298 size_t head = 0;
299
300 SeqHistory() { std::fill(valid, valid + SIZE, false); }
301
302 bool is_duplicate(uint16_t seq) {
303 for (size_t i = 0; i < SIZE; ++i) {
304 if (valid[i] && buffer[i] == seq) return true;
305 }
306 buffer[head] = seq;
307 valid[head] = true;
308 head = (head + 1) % SIZE;
309 return false;
310 }
311 };
312 dmq::xmap<dmq::DelegateRemoteId, SeqHistory> m_history;
313};
