193inline ScopedConnection SignalConnect(std::shared_ptr<SignalState>& state, RecursiveMutex& mutex, const DelegateBase& delegate) {
194 auto copy = std::shared_ptr<DelegateBase>(delegate.Clone(), std::default_delete<DelegateBase>(), ::dmq::stl_allocator<DelegateBase>());
195 if (!copy)
196 BAD_ALLOC();
197
198 dmq::LockGuard<RecursiveMutex> lock(mutex);
199 dmq::LockGuard<RecursiveMutex> stateLock(state->mtx);
200 state->delegates.push_back(copy);
201
202 return ScopedConnection(Connection(
203 std::static_pointer_cast<void>(state),
204 std::static_pointer_cast<void>(copy),

331 static void InvokeSnapshot(const Snapshot& s, Args... args) {
332 if (s.count <= SIGNAL_SBO_COUNT) {
333 for (size_t i = 0; i < s.count; ++i) {
334#if defined(__clang__)
335#pragma clang diagnostic push
336#pragma clang diagnostic ignored "-Wunsafe-buffer-usage"
337#endif
338 if (s.small_buf[i])
339 (*static_cast<DelegateType*>(s.small_buf[i].get()))(args...);
340#if defined(__clang__)
341#pragma clang diagnostic pop
342#endif
343 }
344 } else {
345 for (auto& d : s.large_buf) {
346 if (d)
347 (*static_cast<DelegateType*>(d.get()))(args...);
348 }
349 }
350 }

355 }
