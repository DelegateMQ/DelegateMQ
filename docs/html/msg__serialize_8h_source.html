254 std::istream& read(std::istream& is, I* t_)
255 {
256 if (check_stop_parse(is))
257 return is;
258
259 if (check_pointer(is, t_))
260 {
261 if (read_type(is, Type::USER_DEFINED))
262 {
263 uint16_t size = 0;
264 std::streampos startPos = is.tellg();
265 if (startPos == std::streampos(-1))
266 {
267 raiseError(ParsingError::NON_SEEKABLE_STREAM, __LINE__, __FILE__);
268 is.setstate(std::ios::failbit);
269 return is;
270 }
271
272 read(is, size, false);
273
274 parseStatus(typeid(*t_), size);
275
276 // Save the stop parsing position to prevent parsing overrun
277 push_stop_parse_pos(startPos + std::streampos(size));
278
279 t_->read(*this, is);
280
281 pop_stop_parse_pos();
282
283 if (is.good())
284 {
285 std::streampos endPos = is.tellg();
286 uint16_t rcvdSize = static_cast<uint16_t>(endPos - startPos);
287
288 // Did sender send a larger object than what receiver parsed?
289 if (rcvdSize < size)
290 {
291 // Skip over the extra received data
292 uint16_t seekOffset = size - rcvdSize;
293 is.seekg(seekOffset, std::ios_base::cur);
294 }
295 }
296 return is;
297 }
298 }
299 return is;

427 std::ostream& write(std::ostream& os, I* t_)
428 {
429 if (check_pointer(os, t_))
430 {
431 uint16_t elementSize = 0;
432
433 write_type(os, Type::USER_DEFINED);
434 std::streampos elementSizePos = os.tellp();
435 if (elementSizePos == std::streampos(-1))
436 {
437 raiseError(ParsingError::NON_SEEKABLE_STREAM, __LINE__, __FILE__);
438 os.setstate(std::ios::failbit);
439 return os;
440 }
441 write(os, elementSize, false);
442
443 // Write user defined object
444 t_->write(*this, os);
445
446 if (os.good())
447 {
448 // Backfill the object size into the stream (requires seekable stream)
449 std::streampos currentPos = os.tellp();
450 os.seekp(elementSizePos);
451 elementSize = static_cast<uint16_t>(currentPos - elementSizePos);
452 write(os, elementSize, false);
453 os.seekp(currentPos);
454 }
455 return os;
456 }
457 return os;

1386 std::istream& read_internal(std::istream& is, char* p, uint32_t size, bool no_swap = false)
1387 {
1388 if (check_stop_parse(is))
1389 return is;
1390
1391 if (!check_pointer(is, p))
1392 return is;
1393
1394 // Single read for all bytes, then reverse in-place when host is big-endian
1395 is.read(p, size);
1396 if (!LE() && !no_swap && size > 1)
1397 std::reverse(p, p + size);
1398
1399 return is;
1400 }
1401
1410 std::ostream& write_internal(std::ostream& os, const char* p, uint32_t size, bool no_swap = false)
1411 {
1412 if (!check_pointer(os, p))
1413 return os;
1414
1415 if (!LE() && !no_swap && size > 1)
1416 {
1417 if (size > 8)
1418 {
1419 raiseError(ParsingError::INVALID_INPUT, __LINE__, __FILE__);
1420 os.setstate(std::ios::failbit);
1421 return os;
1422 }
1423
1424 // Reverse bytes into a small stack buffer then write in a single call.
1425 // Primitives are at most 8 bytes (double / int64_t).
1426 char tmp[8];
1427 for (uint32_t i = 0; i < size; ++i)
1428 tmp[i] = p[size - 1 - i];
1429 os.write(tmp, size);
1430 }
1431 else
1432 {
1433 os.write(p, size);
1434 }
1435 return os;
1436 }
