251 std::istream& read(std::istream& is, I* t_)
252 {
253 if (check_stop_parse(is))
254 return is;
255
256 if (check_pointer(is, t_))
257 {
258 if (read_type(is, Type::USER_DEFINED))
259 {
260 uint16_t size = 0;
261 std::streampos startPos = is.tellg();
262 if (startPos == std::streampos(-1))
263 {
264 raiseError(ParsingError::NON_SEEKABLE_STREAM, __LINE__, __FILE__);
265 is.setstate(std::ios::failbit);
266 return is;
267 }
268
269 read(is, size, false);
270
271 parseStatus(typeid(*t_), size);
272
273 // Save the stop parsing position to prevent parsing overrun
274 push_stop_parse_pos(startPos + std::streampos(size));
275
276 t_->read(*this, is);
277
278 pop_stop_parse_pos();
279
280 if (is.good())
281 {
282 std::streampos endPos = is.tellg();
283 uint16_t rcvdSize = static_cast<uint16_t>(endPos - startPos);
284
285 // Did sender send a larger object than what receiver parsed?
286 if (rcvdSize < size)
287 {
288 // Skip over the extra received data
289 uint16_t seekOffset = size - rcvdSize;
290 is.seekg(seekOffset, std::ios_base::cur);
291 }
292 }
293 return is;
294 }
295 }
296 return is;
297 }

424 std::ostream& write(std::ostream& os, I* t_)
425 {
426 if (check_pointer(os, t_))
427 {
428 uint16_t elementSize = 0;
429
430 write_type(os, Type::USER_DEFINED);
431 std::streampos elementSizePos = os.tellp();
432 if (elementSizePos == std::streampos(-1))
433 {
434 raiseError(ParsingError::NON_SEEKABLE_STREAM, __LINE__, __FILE__);
435 os.setstate(std::ios::failbit);
436 return os;
437 }
438 write(os, elementSize, false);
439
440 // Write user defined object
441 t_->write(*this, os);
442
443 if (os.good())
444 {
445 // Backfill the object size into the stream (requires seekable stream)
446 std::streampos currentPos = os.tellp();
447 os.seekp(elementSizePos);
448 elementSize = static_cast<uint16_t>(currentPos - elementSizePos);
449 write(os, elementSize, false);
450 os.seekp(currentPos);
451 }
452 return os;
453 }
454 return os;

574 template<typename T>
575 std::istream& read(std::istream& is, T& t_, bool readPrependedType = true)
576 {
577 static_assert(!serialize_traits::is_unsupported_container<T>::value, "Unsupported C++ container type");
578
579 static_assert(!(std::is_pointer<T>::value &&
580 (std::is_arithmetic<typename std::remove_pointer<T>::type>::value ||
581 std::is_class<typename std::remove_pointer<T>::type>::value)),
582 "T cannot be a pointer to a built-in or custom data type. Use values instead.");
583
584 if (check_stop_parse(is))
585 return is;
586
587 // Is T a built-in data type (e.g. float, int, ...)?
588 if (std::is_class<T>::value == false)
589 {
590 // Is T is not a pointer type
591 if (std::is_pointer<T>::value == false)
592 {
593 if (readPrependedType)
594 {
595 if (!read_type(is, Type::LITERAL))
596 {
597 return is;
598 }
599 }
600
601 if (readPrependedType)
602 parseStatus(typeid(t_), sizeof(t_));
603 read_internal(is, reinterpret_cast<char*>(&t_), sizeof(t_));
604 return is;
605 }
606 else
607 {
608 // Can't read pointers to built-in type
609 raiseError(ParsingError::INVALID_INPUT, __LINE__, __FILE__);
610 is.setstate(std::ios::failbit);
611 return is;
612 }
613 }
614 // Else T is a user defined data type (e.g. MyData)
615 else
616 {
617 // C-style cast required: without if constexpr (C++14) this branch is
618 // compiled for all T including primitives, so static_cast is rejected
619 // for non-class types. The branch is only reached when is_class<T> is
620 // true, making the cast safe at runtime.
621 read(is, (serialize::I*)(&t_));
622 return is;
623 }

642 // types satisfy the first clause so the assertion always passes for them.

644 "T must inherit from serialize::I");

1284 std::istream& read_internal(std::istream& is, char* p, uint32_t size, bool no_swap = false)
1285 {
1286 if (check_stop_parse(is))
1287 return is;
1288
1289 if (!check_pointer(is, p))
1290 return is;
1291
1292 // Single read for all bytes, then reverse in-place when host is big-endian
1293 is.read(p, size);
1294 if (!LE() && !no_swap && size > 1)
1295 std::reverse(p, p + size);
1296
1297 return is;

1308 std::ostream& write_internal(std::ostream& os, const char* p, uint32_t size, bool no_swap = false)
1309 {
1310 if (!check_pointer(os, p))
1311 return os;
1312
1313 if (!LE() && !no_swap && size > 1)
1314 {
1315 if (size > 8)
1316 {
1317 raiseError(ParsingError::INVALID_INPUT, __LINE__, __FILE__);
1318 os.setstate(std::ios::failbit);
1319 return os;
1320 }
1321
1322 // Reverse bytes into a small stack buffer then write in a single call.
1323 // Primitives are at most 8 bytes (double / int64_t).
1324 char tmp[8];
1325 for (uint32_t i = 0; i < size; ++i)
1326 tmp[i] = p[size - 1 - i];
1327 os.write(tmp, size);
1328 }
1329 else
1330 {
1331 os.write(p, size);
1332 }
1333 return os;
1334 }
