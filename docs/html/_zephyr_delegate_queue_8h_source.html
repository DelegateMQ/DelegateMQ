107 ThreadMsg* Receive(k_timeout_t waitOption)
108 {
109 ThreadMsg* msg = nullptr;
110
111 // Fast path: don't pay for k_poll() if something is already pending.
112 if (k_msgq_get(&m_highMsgq, &msg, K_NO_WAIT) == 0)
113 return msg;
114 if (k_msgq_get(&m_normalMsgq, &msg, K_NO_WAIT) == 0)
115 return msg;
116
117 if (K_TIMEOUT_EQ(waitOption, K_NO_WAIT))
118 return nullptr;
119
120 struct k_poll_event events[2];
121 k_poll_event_init(&events[0], K_POLL_TYPE_MSGQ_DATA_AVAILABLE, K_POLL_MODE_NOTIFY_ONLY, &m_highMsgq);
122 k_poll_event_init(&events[1], K_POLL_TYPE_MSGQ_DATA_AVAILABLE, K_POLL_MODE_NOTIFY_ONLY, &m_normalMsgq);
123
124 if (k_poll(events, 2, waitOption) != 0)
125 return nullptr; // timed out waiting for either queue
126
127 // k_poll only signals readiness; re-check high first so a HIGH
128 // message still jumps the FIFO even if NORMAL became ready too.
129 if (k_msgq_get(&m_highMsgq, &msg, K_NO_WAIT) == 0)
130 return msg;
131 if (k_msgq_get(&m_normalMsgq, &msg, K_NO_WAIT) == 0)
132 return msg;
133 return nullptr;
134 }
