34 // Write arguments to a stream
35 virtual std::ostream& Write(std::ostream& os, const Args&... args) override {
36 try {
37 // Reset stream position. DelegateMQ reuses the stream instance.
38 // If we don't reset, we append new data to old data, sending [Old][New]
39 // and the receiver will always deserialize [Old].
40 os.seekp(0, std::ios::beg);
41
42 // Optimization: If it's a stringstream, clear it completely to avoid
43 // sending "tail garbage" if the new packet is smaller than the previous one.
44 // (Note: Even with garbage tail, msgpack works because it stops reading
45 // after the valid object, but clearing saves network bandwidth).
46 auto* ss = dynamic_cast<xostringstream*>(&os);
47 if (ss) {
48 ss->str("");
49 }
50
51 ::msgpack::sbuffer buffer;
52 make_serialized(buffer, args...);
53 os.write(buffer.data(), buffer.size());
54 }
55 catch (const std::exception& e) {
56 std::cerr << "Serialize error: " << e.what() << std::endl;
57 throw;
58 }

60 }

63 virtual std::istream& Read(std::istream& is, Args&... args) override {
64 try {
65 // Read entire stream into memory buffer
66 std::vector<char> buffer_data((std::istreambuf_iterator<char>(is)), std::istreambuf_iterator<char>());
67
68 if (buffer_data.empty() && sizeof...(Args) > 0) {
69 return is;
70 }
71
72 size_t offset = 0;
73
74 // Helper lambda to unpack one argument at a time from the buffer
75 auto unpack_one = [&](auto& arg) {
76 // msgpack::unpack parses one object and updates 'offset' to point to the next byte
77 ::msgpack::object_handle oh = ::msgpack::unpack(buffer_data.data(), buffer_data.size(), offset);
