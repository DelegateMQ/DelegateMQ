63 int Create(Type type, LPCSTR addr, USHORT port)
64 {
65 const std::lock_guard<dmq::RecursiveMutex> lock(m_mutex);
66 m_type = type;
67
68 // Create a UDP socket
69 m_socket = socket(AF_INET, SOCK_DGRAM, IPPROTO_UDP);
70 if (m_socket == INVALID_SOCKET)
71 {
72 std::cerr << "Socket creation failed." << std::endl;
73 return -1;
74 }
75
76 if (type == Type::PUB)
77 {
78 // Set up the server address structure
79 m_addr.sin_family = AF_INET;
80 m_addr.sin_port = htons(port); // Convert port to network byte order
81
82 // Convert IP address string to binary form
83 if (inet_pton(AF_INET, addr, &m_addr.sin_addr) != 1)
84 {
85 std::cerr << "Invalid IP address format: " << addr << std::endl;
86 return -1;
87 }
88
89 DWORD timeout = 2;
90 setsockopt(m_socket, SOL_SOCKET, SO_RCVTIMEO, (const char*)&timeout, sizeof(timeout));
91 }
92 else if (type == Type::SUB)
93 {
94 // Set up the server address structure
95 m_addr.sin_family = AF_INET;
96 m_addr.sin_port = htons(port); // Convert port to network byte order
97 m_addr.sin_addr.s_addr = INADDR_ANY; // Bind to all available interfaces
98
99 // Bind the socket to the address and port
100 int err = ::bind(m_socket, (sockaddr*)&m_addr, sizeof(m_addr));
101 if (err == SOCKET_ERROR)
102 {
103 int wsaErr = WSAGetLastError();
104 std::cerr << "Win32UdpTransport: Bind failed on port " << port << ". Error: " << wsaErr << std::endl;
105 return -1;
106 }
107
108 // Set a 2-second receive timeout
109 DWORD timeout = 2000; // 2000 milliseconds = 2 seconds
110 err = setsockopt(m_socket, SOL_SOCKET, SO_RCVTIMEO, (const char*)&timeout, sizeof(timeout));
111 if (err == SOCKET_ERROR)
112 {
113 std::cerr << "setsockopt(SO_RCVTIMEO) failed: " << WSAGetLastError() << std::endl;
114 return -1;
115 }
116 }
117
118 return 0;

144 virtual int Send(dmq::xostringstream& os, const DmqHeader& header) override
145 {
146 const std::lock_guard<dmq::RecursiveMutex> lock(m_mutex);
147 if (os.bad() || os.fail()) {
148 std::cout << "Error: xostringstream is in a bad state!" << std::endl;
149 return -1;
150 }
151
152 if (m_type == Type::SUB && header.GetId() != dmq::ACK_REMOTE_ID) {
153 std::cout << "Error: Cannot send (non-ACK) on SUB socket!" << std::endl;
154 return -1;
155 }
156
157 if (m_sendTransport != this) {
158 std::cout << "Error: This transport used for receive only!" << std::endl;
159 return -1;
160 }
161
162 // Get payload data without string copy if possible
163 auto payload = os.str();
164 uint16_t payloadLen = static_cast<uint16_t>(payload.length());
165
166 if (payloadLen > (BUFFER_SIZE - DmqHeader::HEADER_SIZE)) {
167 std::cerr << "Error: Payload too large for static buffer." << std::endl;
168 return -1;
169 }
170
171 // Prepare header values in Network Byte Order
172 uint16_t marker = htons(header.GetMarker());
173 uint16_t id = htons(header.GetId());
174 uint16_t seqNum = htons(header.GetSeqNum());
175 uint16_t length = htons(payloadLen);
176
177 // Copy header to start of send buffer
178 memcpy(m_sendBuffer, &marker, 2);
179 memcpy(m_sendBuffer + 2, &id, 2);
180 memcpy(m_sendBuffer + 4, &seqNum, 2);
181 memcpy(m_sendBuffer + 6, &length, 2);
182
183 // Copy payload after header
184 if (payloadLen > 0) {
185 memcpy(m_sendBuffer + DmqHeader::HEADER_SIZE, payload.data(), payloadLen);
186 }
187
188 int totalSize = DmqHeader::HEADER_SIZE + payloadLen;
189
190 int err = sendto(m_socket, m_sendBuffer, totalSize, 0, (sockaddr*)&m_addr, sizeof(m_addr));
191
192 if (err == SOCKET_ERROR) {
193 std::cerr << "Win32UdpTransport: ERROR - sendto failed with " << WSAGetLastError() << std::endl;
194 }
195
196 return (err == SOCKET_ERROR) ? -1 : 0;
197 }

199 virtual int Receive(dmq::xstringstream& is, DmqHeader& header) override
200 {
201 const std::lock_guard<dmq::RecursiveMutex> lock(m_mutex);
202 if (m_recvTransport != this) {
203 std::cout << "Error: This transport used for send only!" << std::endl;
204 return -1;
205 }
206
207 sockaddr_in fromAddr{};
208 int addrLen = sizeof(fromAddr);
209 int size = recvfrom(m_socket, m_buffer, sizeof(m_buffer), 0, (sockaddr*)&fromAddr, &addrLen);
210 if (m_type == Type::SUB)
211 m_addr = fromAddr;
212 if (size == SOCKET_ERROR || size < static_cast<int>(DmqHeader::HEADER_SIZE))
213 {
214 return -1;
215 }
216
217 // 1. Read Header directly from m_buffer (Network Byte Order)
218 uint16_t marker, id, seqNum, length;
219 memcpy(&marker, m_buffer, 2);
220 memcpy(&id, m_buffer + 2, 2);
221 memcpy(&seqNum, m_buffer + 4, 2);
222 memcpy(&length, m_buffer + 6, 2);
223
224 header.SetMarker(ntohs(marker));
225 header.SetId(ntohs(id));
226 header.SetSeqNum(ntohs(seqNum));
227 header.SetLength(ntohs(length));
228
229 if (header.GetMarker() != DmqHeader::MARKER) {
230 return -1;
231 }
232
233 // 2. Write payload to provided stream
234 uint16_t payloadSize = header.GetLength();
235 if (payloadSize > (size - DmqHeader::HEADER_SIZE))
236 payloadSize = static_cast<uint16_t>(size - DmqHeader::HEADER_SIZE);
237
238 if (payloadSize > 0) {
239 is.write(m_buffer + DmqHeader::HEADER_SIZE, payloadSize);
240 }
241
242 // 3. Handle Acknowledgment
243 if (header.GetId() == dmq::ACK_REMOTE_ID)
244 {
245 if (m_transportMonitor)
246 m_transportMonitor->Remove(header.GetSeqNum());
247 }
248 else if (m_sendTransport && m_type == Type::SUB)
249 {
250 // Send ACK using a small stack buffer to avoid any heap
251 uint16_t a_marker = htons(DmqHeader::MARKER);
252 uint16_t a_id = htons(dmq::ACK_REMOTE_ID);
253 uint16_t a_seqNum = htons(header.GetSeqNum());
254 uint16_t a_length = 0;
255
256 char ackBuf[DmqHeader::HEADER_SIZE] = {0};
257 memcpy(ackBuf, &a_marker, 2);
258 memcpy(ackBuf + 2, &a_id, 2);
259 memcpy(ackBuf + 4, &a_seqNum, 2);
260 memcpy(ackBuf + 6, &a_length, 2);
261
262 sockaddr_in targetAddr = (m_type == Type::SUB) ? fromAddr : m_addr;
263 sendto(m_socket, ackBuf, static_cast<int>(DmqHeader::HEADER_SIZE), 0, (sockaddr*)&targetAddr, sizeof(targetAddr));
264 }
265
266 return 0;
267 }
