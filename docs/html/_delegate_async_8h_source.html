147template <class... Args>
148void DispatchAsync(std::shared_ptr<IThreadInvoker> invoker, IThread* thread, Priority priority, Args&&... args) {
149 // Create a new message instance for sending to the destination thread. Erase to
150 // the non-templated DelegateMsg base BEFORE constructing the shared_ptr (rather
151 // than xmake_shared<DelegateAsyncMsg<Args...>>) so the shared_ptr control block
152 // is not templated on Args... -- one control-block type is then shared across
153 // every DelegateAsync signature instead of duplicated per signature.
154 auto* rawMsg = new(std::nothrow) DelegateAsyncMsg<Args...>(std::move(invoker), priority, std::forward<Args>(args)...);
155 if (!rawMsg)
156 BAD_ALLOC();
157 DelegateMsg* msgBase = rawMsg;
158 std::shared_ptr<DelegateMsg> msg(msgBase, std::default_delete<DelegateMsg>(), dmq::stl_allocator<DelegateMsg>());
159
160 if (thread) {
161 // Dispatch message onto the callback destination thread. Invoke()
162 // will be called by the destintation thread.
163 bool success = thread->DispatchDelegate(msg);
164 if (!success) {
165 LOG_ERROR("DelegateAsync dispatch failed");
166 }
167 }
168}
