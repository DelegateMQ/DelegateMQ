27 // Helper class to implement QoS rate limiting without lambda captures.
28 template <typename T>
29 class RateLimiter {
30 XALLOCATOR
31 public:
32 RateLimiter(dmq::UnicastDelegate<void(const T&)> func, uint32_t minSepRep)
33 : m_func(std::move(func)), m_minSepRep(minSepRep), m_lastDeliveryRep(0) {}
34
35 void Invoke(const T& data) {
36 auto nowRep = static_cast<uint32_t>(std::chrono::duration_cast<std::chrono::milliseconds>(dmq::Clock::now().time_since_epoch()).count());
37 auto lastRep = m_lastDeliveryRep.load(std::memory_order_relaxed);
38 if (nowRep - lastRep >= m_minSepRep) {
39 m_lastDeliveryRep.store(nowRep, std::memory_order_relaxed);
40 m_func(data);
41 }
42 }
43
44 private:
45 dmq::UnicastDelegate<void(const T&)> m_func;
46 uint32_t m_minSepRep;
47 std::atomic<uint32_t> m_lastDeliveryRep;
48 };

106 template <typename T, typename F, typename P>
107 [[nodiscard]] static dmq::ScopedConnection SubscribeFilter(const dmq::xstring& topic, F&& func, P&& predicate, dmq::IThread* thread = nullptr, QoS qos = {}) {
108 dmq::UnicastDelegate<void(const T&)> funcUD;
109 if constexpr (std::is_base_of_v<dmq::Delegate<void(const T&)>, std::decay_t<F>>)

274 [[nodiscard]] dmq::ScopedConnection InternalSubscribe(const dmq::xstring& topic, dmq::UnicastDelegate<void(const T&)> typedFunc, dmq::IThread* thread, QoS qos) {
275 SignalPtr<T> signal;
276
277 // Wrap with min separation rate limiter if requested. Each subscriber gets its
278 // own independent last-delivery timestamp, so different subscribers on the same
279 // topic can have different (or no) rate limits without affecting each other.
280 if (qos.minSeparation.has_value()) {
281 auto minSepRep = static_cast<uint32_t>(std::chrono::duration_cast<std::chrono::milliseconds>(qos.minSeparation.value()).count());
282 auto limiter = dmq::xmake_shared<detail::RateLimiter<T>>(std::move(typedFunc), minSepRep);
283 // Capture limiter by shared_ptr so the RateLimiter stays alive for the connection's lifetime.
284 // MakeDelegate(limiter.get(), ...) would store a raw pointer; the shared_ptr must be in the closure.
285 typedFunc = dmq::DelegateFunction<void(const T&)>([limiter](const T& data) { limiter->Invoke(data); });
286 }
287
288 T* cachedValPtr = nullptr;
289 T cachedVal;
290 dmq::ScopedConnection conn;
291
292 {
293 std::lock_guard<dmq::RecursiveMutex> lock(m_mutex);
294
295 // 1. Enable LVC if requested (persists for topic lifetime until ResetForTesting)
296 if (qos.lastValueCache) {
297 m_topicQos[topic].lastValueCache = true;
298 }
299
300 // 2. Get or create signal with type safety check (std::type_index)
301 signal = GetOrCreateSignal<T>(topic);
302 }
303
304 if (!signal) {
305 return {}; // Type mismatch or other failure
306 }
307
308 // 3. Establish connection OUTSIDE the lock to prevent deadlock with Timer/Signal locks.
309 // NOTE: There is a theoretical race where a publish happens between releasing the
310 // DataBus lock and acquiring the Signal lock. However, both use RecursiveMutex
311 // and InternalPublish also snapshots signals outside its lock, so this is
312 // architecturally consistent with the "lock-free dispatch" pattern used elsewhere.
313 using AsyncDelType = dmq::DelegateFunctionAsync<void(const T&)>;
314 std::optional<AsyncDelType> asyncDel;
315 if (thread) {
316 asyncDel.emplace(dmq::MakeDelegate(std::function<void(const T&)>(std::move(typedFunc)), *thread));
317 conn = signal->Connect(*asyncDel);
318 } else {
319 conn = signal->Connect(typedFunc);
320 }
321
322 {
323 std::lock_guard<dmq::RecursiveMutex> lock(m_mutex);
324 // 4. Prepare LVC delivery if enabled and available
325 if (qos.lastValueCache) {
326 auto it = m_lastValues.find(topic);
327 if (it != m_lastValues.end()) {
328 // Check lifespan: skip delivery if the cached value is too old
329 bool expired = false;
330 if (qos.lifespan.has_value()) {
331 auto age = dmq::Clock::now() - it->second.timestamp;
332 expired = (age > qos.lifespan.value());
333 }
334 if (!expired) {
335 cachedVal = *std::static_pointer_cast<T>(it->second.value);
336 cachedValPtr = &cachedVal;
337 }
338 }
339 }
340 }
341
342 // 5. Dispatch LVC outside the lock to prevent deadlocks.
343 // IMPORTANT: Because this happens after releasing the lock, a high-frequency
344 // publisher on another thread could have already sent a new value to the
345 // connected signal. The subscriber might receive the fresh value FIRST,
346 // followed by this stale LVC value — a "rewind" to an older state.
347 //
348 // Recommended mitigation: embed a monotonic sequence number in every message
349 // (see MessageBase) and reject stale arrivals in the subscriber using
350 // dmq::util::MonotonicGuard::IsNewer(). This is an application-level guard
351 // and is the correct fix — resolving the race inside the DataBus lock would
352 // require holding the lock across the async dispatch, which deadlocks.
353 if (cachedValPtr) {
354 if (asyncDel)
355 asyncDel->AsyncInvoke(*cachedValPtr);
356 else
357 typedFunc(*cachedValPtr);
358 }
359
360 return conn;
361 }

363 template <typename T>
364 void InternalPublish(const dmq::xstring& topic, const T& data, bool localOnly) {
365 // Capture timestamp before lock acquisition for maximum accuracy and
366 // monotonic ordering using dmq::Clock.
367 auto now = dmq::Clock::now();
368 uint64_t timestamp = std::chrono::duration_cast<std::chrono::microseconds>(now.time_since_epoch()).count();
369
370 SignalPtr<T> signal;
371 std::shared_ptr<void> serializerPtr;
372 dmq::ISerializer<void(T)>* serializer = nullptr;
373 std::array<std::shared_ptr<Participant>, dmq::MAX_PARTICIPANTS> participantsSnapshot;
374 size_t participantSnapshotCount = 0;
375 dmq::xstring strVal = "?";
376 bool hasMonitor = false;
377
378 {
379 std::lock_guard<dmq::RecursiveMutex> lock(m_mutex);
380
381 // 1. Type safety: verify T matches the registered type for this topic.
382 // Must be first — before any writes — so a mismatch never corrupts LVC.
383 auto itType = m_typeIndices.find(topic);
384 if (itType != m_typeIndices.end()) {
385 ASSERT_TRUE(itType->second == std::type_index(typeid(T)));
386 }
387
388 // 2. Update LVC ONLY if enabled for this topic to save memory.
389 // NOTE: QoS lastValueCache is currently "sticky" per topic. Once enabled
390 // by any subscriber, it remains active for that topic until ResetForTesting().
391 auto itQos = m_topicQos.find(topic);
392 if (itQos != m_topicQos.end() && itQos->second.lastValueCache) {
393 m_lastValues[topic] = LvcEntry{ dmq::xmake_shared<T>(data), now };
394 }
395
396 // 3. Prepare monitor data
397 if (!m_monitorSignal.Empty()) {
398 hasMonitor = true;
399 auto itStr = m_stringifiers.find(topic);
400 if (itStr != m_stringifiers.end()) {
401 auto func = static_cast<dmq::UnicastDelegate<dmq::xstring(const T&)>*>(itStr->second.get());
402 strVal = (*func)(data);
403 }
404 }
405
406 // 4. Get signal and remote info. Only create Signal if there is local interest.
407 auto itSig = m_signals.find(topic);
408 if (itSig != m_signals.end()) {
409 signal = std::static_pointer_cast<dmq::Signal<void(const T&)>>(itSig->second);
410 }
411
412 auto itSer = m_serializers.find(topic);
413 if (itSer != m_serializers.end()) {
414 serializerPtr = itSer->second;
415 serializer = static_cast<dmq::ISerializer<void(T)>*>(serializerPtr.get());
416 }
417
418 // 5. Snapshot participants while locked to ensure atomicity between
419 // local and remote dispatch sets.
420 for (size_t i = 0; i < m_participantCount; ++i)
421 participantsSnapshot[i] = m_participants[i];
422 participantSnapshotCount = m_participantCount;
423 }
424
425 // 6. Dispatch Monitor outside lock to allow re-entry/prevent deadlocks
426 if (hasMonitor) {
427 SpyPacket packet{ topic, strVal, timestamp };
428 m_monitorSignal(packet);
429 }
430
431 // 7. Local distribution
432 bool handled = false;
433 if (signal) {
434 (*signal)(data);
435 handled = true;
436 }
437
438 // 8. Remote distribution using the snapshot
439 if (!localOnly) {
440 Participant* interested[dmq::MAX_PARTICIPANTS];
441 size_t interestedCount = 0;
442 for (size_t i = 0; i < participantSnapshotCount; ++i) {
443 dmq::DelegateRemoteId rid;
444 if (participantsSnapshot[i]->GetRemoteId(topic, rid)) {
445 interested[interestedCount++] = participantsSnapshot[i].get();
446 }
447 }
448
449 if (interestedCount > 0) {
450 if (serializer) {
451 for (size_t i = 0; i < interestedCount; ++i) {
452 interested[i]->Send<T>(topic, data, *serializer);
453 handled = true;
454 }
455 } else {
456 // Topic has remote interest but no serializer — fire once per topic.
457 bool shouldFire = false;
458 {
459 std::lock_guard<dmq::RecursiveMutex> lock(m_mutex);
460 constexpr uint8_t bit = uint8_t(1u << static_cast<int>(dmq::DelegateError::ERR_NO_SERIALIZER));
461 auto& bits = m_reportedErrors[topic];
462 if (!(bits & bit)) { bits |= bit; shouldFire = true; }
463 }
464 if (shouldFire)
465 m_errorSignal(topic, dmq::DelegateError::ERR_NO_SERIALIZER);
466 handled = true;
467 }
468 }
469 }
470
471 // 9. Notify if no one received the message
472 if (!handled) {
473 m_unhandledSignal(topic);
474 }
475 }

547 if (itType != m_typeIndices.end()) {
